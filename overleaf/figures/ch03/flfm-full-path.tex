% The full FLFM light path with both relays (supervisor, 2026-11-23
% meeting), shown at the end of Chapter 3 after the idealized path of
% flfm-simple-path.tex. The back pupil lies inside the objective, so the tube
% lens and Fourier lens relay it onto the microlens array; a camera relay maps
% the lenslet images onto the sensor. The plane between the tube lens and the
% Fourier lens is left unlabeled (rule 7). Five lenslets drawn; not to scale.
\begin{tikzpicture}[
    x=1cm,y=1cm,
    font=\footnotesize,
    ink/.style={text=black!85},
    lens/.style={<->,>={Stealth[length=2mm]},draw=black!75,line width=0.8pt},
    ray/.style={draw=blue!60!black,line width=0.7pt},
    relayray/.style={draw=blue!45,line width=0.5pt},
    slice/.style={draw=black!25,line width=0.4pt},
    plane/.style={draw=black!45,dashed,line width=0.5pt},
    subimg/.style={draw=orange!75!black,line width=2.2pt,opacity=0.55},
    dim/.style={<->,>={Stealth[length=1.6mm]},draw=black!70,line width=0.5pt},
    bracket/.style={draw=black!60,line width=0.5pt},
    lab/.style={font=\footnotesize,ink,inner sep=1.5pt,align=center}
]
% ---- geometry (cm; not to scale) ----
% object 0 | objective 1.0 | back pupil 2.0 | tube lens 3.6 | image 5.2
% | Fourier lens 7.6 | MLA 10.0 | MLA focal plane 11.0
% | relay lens 1 12.35 | relay Fourier plane 13.7 | relay lens 2 14.6 | sensor 15.5
\def\xo{1.0}  \def\xp{2.0}  \def\xt{3.6}  \def\xn{5.2}
\def\xf{7.6}  \def\xm{10.0} \def\xq{11.0}
\def\fa{1.35} \def\fb{0.9}
\pgfmathsetmacro{\xa}{\xq+\fa}
\pgfmathsetmacro{\xb}{\xa+\fa+\fb}
\pgfmathsetmacro{\xs}{\xb+\fb}
\pgfmathsetmacro{\mc}{\fb/\fa}          % camera relay magnification (schematic)
\def\hp{0.75}                           % half of D_pupil
\def\hq{1.29}                           % half of D_proj (= hp f_FL/f_TL)
\def\dm{0.516}                          % lenslet pitch d_MLA (5 across D_proj)
\def\sfr{0.8}                           % image of the field, fraction of one lenslet pitch

% optical axis
\draw[draw=black!30,line width=0.4pt] (-0.4,0) -- (\xs+0.5,0);

% planes: object, back pupil, the (unnamed) image plane, MLA focal plane
\draw[plane] (0,-1.5) -- (0,1.5);
\draw[plane] (\xp,-1.5) -- (\xp,1.5);
\draw[plane] (\xn,-1.5) -- (\xn,1.5);
\draw[plane] (\xq,-1.5) -- (\xq,1.5);

% the objective's barrel encloses its back pupil
\draw[draw=black!35,fill=black!4,line width=0.4pt,rounded corners=2pt]
    (\xo-0.25,-1.15) rectangle (\xp+0.2,1.15);

% lenses
\draw[lens] (\xo,-1.05) -- (\xo,1.05);
\draw[lens] (\xt,-1.15) -- (\xt,1.15);
\draw[lens] (\xf,-1.6) -- (\xf,1.6);
\draw[lens] (\xa,-1.45) -- (\xa,1.45);
\draw[lens] (\xb,-1.0) -- (\xb,1.0);

% microlens array: 5 lenslets across the pupil image
\foreach \k in {-2,...,2} {
    \draw[draw=black!70,fill=black!6,line width=0.5pt]
        (\xm-0.07,{\k*\dm-0.5*\dm}) rectangle (\xm+0.07,{\k*\dm+0.5*\dm});
}

% rays of one on-axis emitter (marginal rays)
\draw[ray] (0,0) -- (\xo,\hp);
\draw[ray] (0,0) -- (\xo,-\hp);
\draw[ray] (\xo,\hp) -- (\xt,\hp);
\draw[ray] (\xo,-\hp) -- (\xt,-\hp);
\draw[ray] (\xt,\hp) -- (\xn,0);
\draw[ray] (\xt,-\hp) -- (\xn,0);
\draw[ray] (\xn,0) -- (\xf,\hq);
\draw[ray] (\xn,0) -- (\xf,-\hq);
\draw[ray] (\xf,\hq) -- (\xm,\hq);
\draw[ray] (\xf,-\hq) -- (\xm,-\hq);
\foreach \k in {-2,...,2} {
    \draw[slice] (\xf,{\k*\dm-0.5*\dm}) -- (\xm,{\k*\dm-0.5*\dm});
    \draw[ray] (\xm,{\k*\dm-0.5*\dm}) -- (\xq,{\k*\dm});
    \draw[ray] (\xm,{\k*\dm+0.5*\dm}) -- (\xq,{\k*\dm});
}
\draw[slice] (\xf,{2.5*\dm}) -- (\xm,{2.5*\dm});

% views in the MLA focal plane
\foreach \k in {-2,...,2} {
    \draw[subimg] (\xq,{\k*\dm-0.5*\sfr*\dm}) -- (\xq,{\k*\dm+0.5*\sfr*\dm});
    \fill[orange!80!black] (\xq,{\k*\dm}) circle (0.045);
}

% camera relay (4f): each view is imaged, inverted, onto the sensor
\foreach \k in {-2,...,2} {
    \draw[relayray] (\xq,{\k*\dm}) -- (\xa,{\k*\dm}) -- (\xa+\fa,0)
        -- (\xb,{-\k*\dm*\mc}) -- (\xs,{-\k*\dm*\mc});
}

% sensor with pixel ticks, and the views on it
\draw[draw=black!70,fill=black!12,line width=0.5pt] (\xs,-1.0) rectangle (\xs+0.12,1.0);
\foreach \y in {-0.96,-0.92,...,0.97} {
    \draw[draw=black!45,line width=0.25pt] (\xs,\y) -- (\xs+0.12,\y);
}
\foreach \k in {-2,...,2} {
    \draw[subimg] (\xs-0.02,{-\k*\dm*\mc-0.5*\sfr*\dm*\mc})
        -- (\xs-0.02,{-\k*\dm*\mc+0.5*\sfr*\dm*\mc});
}

% dimensions
\draw[dim] (\xp+0.45,-\hp) -- (\xp+0.45,\hp);
\node[lab,anchor=west,fill=white,inner sep=1pt] at (\xp+0.5,-0.35) {$D_{pupil}$};
\draw[dim] (8.8,-\hq) -- (8.8,\hq);
\node[lab,anchor=east,fill=white,inner sep=1pt] at (8.75,0.0) {$D_{proj}$};
\draw[dim] (\xm-0.4,{1.5*\dm}) -- (\xm-0.4,{2.5*\dm});
\node[lab,anchor=east,fill=white,inner sep=1pt] at (\xm-0.45,{2.0*\dm}) {$d_{MLA}$};
\draw[dim] (\xs+0.35,-1.0) -- (\xs+0.35,1.0);
\node[lab,anchor=west,inner sep=1pt] at (\xs+0.4,0) {$D_{cam}$};

% focal lengths along the bottom
\def\yb{-2.85}
\draw[dim] (0,\yb) -- (\xo,\yb);        \node[lab,fill=white] at ({0.5*\xo},\yb) {$f_{obj}$};
\draw[dim] (\xp,\yb) -- (\xt,\yb);      \node[lab,fill=white] at ({0.5*(\xp+\xt)},\yb) {$f_{TL}$};
\draw[dim] (\xt,\yb) -- (\xn,\yb);      \node[lab,fill=white] at ({0.5*(\xt+\xn)},\yb) {$f_{TL}$};
\draw[dim] (\xn,\yb) -- (\xf,\yb);      \node[lab,fill=white] at ({0.5*(\xn+\xf)},\yb) {$f_{FL}$};
\draw[dim] (\xf,\yb) -- (\xm,\yb);      \node[lab,fill=white] at ({0.5*(\xf+\xm)},\yb) {$f_{FL}$};
\draw[dim] (\xm,\yb) -- (\xq,\yb);      \node[lab,fill=white] at ({0.5*(\xm+\xq)},\yb) {$f_{MLA}$};
\draw[dim] (\xq,\yb) -- (\xa,\yb);      \node[lab,fill=white] at ({0.5*(\xq+\xa)},\yb) {$f_{c1}$};
\draw[dim] (\xa,\yb) -- (\xa+\fa,\yb);  \node[lab,fill=white] at ({\xa+0.5*\fa},\yb) {$f_{c1}$};
\draw[dim] (\xa+\fa,\yb) -- (\xb,\yb);  \node[lab,fill=white] at ({\xa+\fa+0.5*\fb},\yb) {$f_{c2}$};
\draw[dim] (\xb,\yb) -- (\xs,\yb);      \node[lab,fill=white] at ({\xb+0.5*\fb},\yb) {$f_{c2}$};

% names
\node[lab,anchor=south] at (0,1.55) {object\\plane};
\node[lab,anchor=south] at (\xo,1.2) {objective};
\node[lab,anchor=north] at (\xp,-1.55) {back pupil\\(inside the\\objective)};
\node[lab,anchor=south] at (\xt,1.2) {tube lens\\(TL)};
\node[lab,anchor=south] at (\xf,1.65) {Fourier lens\\(FL)};
\node[lab,anchor=south] at (\xm,1.65) {MLA};
\node[lab,anchor=north] at (\xq,-1.55) {views};
\node[lab,anchor=south] at (\xs+0.06,1.65) {sensor};

% relay brackets
\draw[bracket] (\xt-0.1,-1.75) -- (\xt-0.1,-1.85) -- (\xf+0.1,-1.85) -- (\xf+0.1,-1.75);
\node[lab,anchor=north] at ({0.5*(\xt+\xf)},-1.87) {pupil relay, $\times f_{FL}/f_{TL}$};
\draw[bracket] (\xa-0.1,1.52) -- (\xa-0.1,1.62) -- (\xb+0.1,1.62) -- (\xb+0.1,1.52);
\node[lab,anchor=south] at ({0.5*(\xa+\xb)},1.64) {camera relay\\$M_{cam}=f_{c2}/f_{c1}$};
\end{tikzpicture}
